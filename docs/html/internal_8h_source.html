42typedef int64_t aaru_off_t;
43
44/* Seek using a 64-bit offset. Returns 0 on success, non-zero on failure, just
45 * like fseek()/_fseeki64(). Use aaruf_ftell() to retrieve the current offset. */
46static inline int aaruf_fseek(FILE *stream, aaru_off_t offset, int origin)
47{
48 return AARU_FSEEK(stream, offset, origin);
49}
50
51/* Return the current file offset as a signed 64-bit value. */
52static inline aaru_off_t aaruf_ftell(FILE *stream)
53{
54 return (aaru_off_t)AARU_FTELL(stream);
55}

59#define ftell(stream) aaruf_ftell((stream))
60#endif
