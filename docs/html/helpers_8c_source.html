463int compare_extents(const void *a, const void *b)
464{
465 const DumpExtent *extent_a = a;
466 const DumpExtent *extent_b = b;
467
468 if(extent_a->start < extent_b->start) return -1;
469 if(extent_a->start > extent_b->start) return 1;
470 return 0;
471}

484void generate_random_bytes(uint8_t *buffer, size_t length)
485{
486 if(buffer == NULL || length == 0) return;
487
488#if defined(_WIN32) || defined(_WIN64)
489 // Windows implementation using CryptGenRandom
490 HCRYPTPROV hCryptProv;
491 if(CryptAcquireContext(&hCryptProv, NULL, NULL, PROV_RSA_FULL, CRYPT_VERIFYCONTEXT))
492 {
493 CryptGenRandom(hCryptProv, (DWORD)length, buffer);
494 CryptReleaseContext(hCryptProv, 0);
495 return;
496 }
497#else
498 // Unix-like systems (Linux, macOS, BSD, etc.) - use /dev/urandom
499 FILE *urandom = fopen("/dev/urandom", "rb");
500 if(urandom != NULL)
501 {
502 size_t bytes_read = fread(buffer, 1, length, urandom);
503 fclose(urandom);
504 if(bytes_read == length) return;
505 }
506#endif
507
508 // Fallback: use time-seeded rand() if platform-specific methods fail
509 // This is less secure but ensures we always generate some random data
510 srand((unsigned int)time(NULL));
511 for(size_t i = 0; i < length; i++) { buffer[i] = (uint8_t)(rand() % 256); }
512}
