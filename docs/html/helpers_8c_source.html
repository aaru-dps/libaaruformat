511int compare_extents(const void *a, const void *b)
512{
513 const DumpExtent *extent_a = a;
514 const DumpExtent *extent_b = b;
515
516 if(extent_a->start < extent_b->start) return -1;
517 if(extent_a->start > extent_b->start) return 1;
518 return 0;
519}

532void generate_random_bytes(uint8_t *buffer, size_t length)
533{
534 if(buffer == NULL || length == 0) return;
535
536#if defined(_WIN32) || defined(_WIN64)
537 // Windows implementation using CryptGenRandom
538 HCRYPTPROV hCryptProv;
539 if(CryptAcquireContext(&hCryptProv, NULL, NULL, PROV_RSA_FULL, CRYPT_VERIFYCONTEXT))
540 {
541 CryptGenRandom(hCryptProv, (DWORD)length, buffer);
542 CryptReleaseContext(hCryptProv, 0);
543 return;
544 }
545#else
546 // Unix-like systems (Linux, macOS, BSD, etc.) - use /dev/urandom
547 FILE *urandom = fopen("/dev/urandom", "rb");
548 if(urandom != NULL)
549 {
550 size_t bytes_read = fread(buffer, 1, length, urandom);
551 fclose(urandom);
552 if(bytes_read == length) return;
553 }
554#endif
555
556 // Fallback: use time-seeded rand() if platform-specific methods fail
557 // This is less secure but ensures we always generate some random data
558 srand((unsigned int)time(NULL));
559 for(size_t i = 0; i < length; i++) { buffer[i] = (uint8_t)(rand() % 256); }
560}
