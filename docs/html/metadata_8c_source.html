104AARU_EXPORT int32_t AARU_CALL aaruf_get_geometry(const void *context, uint32_t *cylinders, uint32_t *heads,
105 uint32_t *sectors_per_track)
106{
107 TRACE("Entering aaruf_get_geometry(%p, %p, %p, %p)", context, cylinders, heads, sectors_per_track);
108
109 const aaruformat_context *ctx = NULL;
110
111 if(context == NULL)
112 {
113 FATAL("Invalid context");
114
115 TRACE("Exiting aaruf_get_geometry() = AARUF_ERROR_NOT_AARUFORMAT");
116 return AARUF_ERROR_NOT_AARUFORMAT;
117 }
118
119 ctx = context;
120
121 // Not a libaaruformat context
122 if(ctx->magic != AARU_MAGIC)
123 {
124 FATAL("Invalid context");
125
126 TRACE("Exiting aaruf_get_geometry() = AARUF_ERROR_NOT_AARUFORMAT");
127 return AARUF_ERROR_NOT_AARUFORMAT;
128 }
129
130 if(ctx->geometry_block.identifier != GeometryBlock)
131 {
132 FATAL("No geometry block present");
133
134 TRACE("Exiting aaruf_get_geometry() = AARUF_ERROR_CANNOT_READ_BLOCK");
135 return AARUF_ERROR_CANNOT_READ_BLOCK;
136 }
137
138 *cylinders = ctx->geometry_block.cylinders;
139 *heads = ctx->geometry_block.heads;
140 *sectors_per_track = ctx->geometry_block.sectorsPerTrack;
141
142 TRACE("Exiting aaruf_get_geometry(%p, %u, %u, %u) = AARUF_STATUS_OK", context, *cylinders, *heads,
143 *sectors_per_track);
144 return AARUF_STATUS_OK;
145}
