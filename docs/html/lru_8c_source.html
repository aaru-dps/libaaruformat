68void add_to_cache_uint64(struct CacheHeader *cache, const uint64_t key, void *value, const size_t size)
69{
70 struct CacheEntry *entry = NULL;
71
72 // Replace an existing entry for this key. A blind HASH_ADD would leave the old entry in the
73 // table unreachable by HASH_FIND, and its value alive until the whole cache is freed.
74 HASH_FIND(hh, cache->cache, &key, sizeof(uint64_t), entry);
75
76 if(entry)
77 {
78 cache->cur_bytes -= entry->size;
79
80 if(cache->free_func && entry->value && entry->value != value) cache->free_func(entry->value);
81
82 HASH_DELETE(hh, cache->cache, entry);
83 }
84 else
85 {
86 entry = malloc(sizeof(struct CacheEntry));
87 if(!entry) return;
88 }
89
90 entry->key = key;
91 entry->value = value;
92 entry->size = size;
93 HASH_ADD(hh, cache->cache, key, sizeof(uint64_t), entry);
94 cache->cur_bytes += size;
95
96 if(cache->max_bytes == 0) return;
97
98 // Evict least recently used entries until back under budget. Never evict the entry just
99 // inserted: the caller still uses that pointer after this returns.
100 while(cache->cur_bytes > cache->max_bytes && HASH_COUNT(cache->cache) > 1)
101 {
102 struct CacheEntry *oldest = cache->cache;
103
104 if(oldest == NULL || oldest == entry) break;
105
106 drop_entry(cache, oldest);
107 }
108}
