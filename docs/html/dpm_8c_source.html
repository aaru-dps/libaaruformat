43AARU_EXPORT int32_t AARU_CALL aaruf_get_dpm(const void *context, uint8_t *buffer, size_t *length)
44{
45 TRACE("Entering aaruf_get_dpm(%p, %p, %p)", context, buffer, length);
46
47 if(context == NULL || length == NULL)
48 {
49 FATAL("Invalid context");
50
51 TRACE("Exiting aaruf_get_dpm() = AARUF_ERROR_NOT_AARUFORMAT");
52 return AARUF_ERROR_NOT_AARUFORMAT;
53 }
54
55 const aaruformat_context *ctx = context;
56
57 if(ctx->magic != AARU_MAGIC)
58 {
59 FATAL("Invalid context");
60
61 TRACE("Exiting aaruf_get_dpm() = AARUF_ERROR_NOT_AARUFORMAT");
62 return AARUF_ERROR_NOT_AARUFORMAT;
63 }
64
65 if(ctx->dpm_block == NULL || ctx->dpm_block_length == 0)
66 {
67 TRACE("No DPM present");
68 *length = 0;
69
70 TRACE("Exiting aaruf_get_dpm() = AARUF_ERROR_METADATA_NOT_PRESENT");
71 return AARUF_ERROR_METADATA_NOT_PRESENT;
72 }
73
74 if(buffer == NULL || *length < ctx->dpm_block_length)
75 {
76 *length = ctx->dpm_block_length;
77
78 TRACE("Exiting aaruf_get_dpm() = AARUF_ERROR_BUFFER_TOO_SMALL");
79 return AARUF_ERROR_BUFFER_TOO_SMALL;
80 }
81
82 memcpy(buffer, ctx->dpm_block, ctx->dpm_block_length);
83 *length = ctx->dpm_block_length;
84
85 TRACE("Exiting aaruf_get_dpm() = AARUF_STATUS_OK");
86 return AARUF_STATUS_OK;
87}

102AARU_EXPORT int32_t AARU_CALL aaruf_set_dpm(void *context, const uint8_t *data, size_t length)
103{
104 TRACE("Entering aaruf_set_dpm(%p, %p, %zu)", context, data, length);
105
106 if(context == NULL)
107 {
108 FATAL("Invalid context");
109
110 TRACE("Exiting aaruf_set_dpm() = AARUF_ERROR_NOT_AARUFORMAT");
111 return AARUF_ERROR_NOT_AARUFORMAT;
112 }
113
114 aaruformat_context *ctx = context;
115
116 if(ctx->magic != AARU_MAGIC)
117 {
118 FATAL("Invalid context");
119
120 TRACE("Exiting aaruf_set_dpm() = AARUF_ERROR_NOT_AARUFORMAT");
121 return AARUF_ERROR_NOT_AARUFORMAT;
122 }
123
124 if(!ctx->is_writing)
125 {
126 FATAL("Trying to write a read-only image");
127
128 TRACE("Exiting aaruf_set_dpm() = AARUF_READ_ONLY");
129 return AARUF_READ_ONLY;
130 }
131
132 const int32_t status = validate_dpm_block(data, length);
133
134 if(status != AARUF_STATUS_OK)
135 {
136 FATAL("Invalid DPM block");
137
138 TRACE("Exiting aaruf_set_dpm() = %d", status);
139 return status;
140 }
141
142 uint8_t *copy = malloc(length);
143
144 if(copy == NULL)
145 {
146 FATAL("Could not allocate memory for DPM block");
147
148 TRACE("Exiting aaruf_set_dpm() = AARUF_ERROR_NOT_ENOUGH_MEMORY");
149 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
150 }
151
152 memcpy(copy, data, length);
153 free(ctx->dpm_block);
154 ctx->dpm_block = copy;
155 ctx->dpm_block_length = length;
156 ctx->dirty_dpm_block = true;
157
158 TRACE("Exiting aaruf_set_dpm() = AARUF_STATUS_OK");
159 return AARUF_STATUS_OK;
160}
