53int32_t validate_dpm_block(const uint8_t *data, size_t length)
54{
55 if(data == NULL || length < sizeof(DpmHeader)) return AARUF_ERROR_INCORRECT_DATA_SIZE;
56
57 DpmHeader header;
58 memcpy(&header, data, sizeof(DpmHeader));
59
60 if(header.identifier != DataPositionMeasurementBlock) return AARUF_ERROR_CANNOT_READ_BLOCK;
61
62 if(header.length != length - sizeof(DpmHeader) || header.length != dpm_payload_length(&header) ||
63 header.length > UINT32_MAX)
64 return AARUF_ERROR_INCORRECT_DATA_SIZE;
65
66 if(aaruf_crc64_data(data + sizeof(DpmHeader), (uint32_t)header.length) != header.crc64)
67 return AARUF_ERROR_INVALID_BLOCK_CRC;
68
69 return AARUF_STATUS_OK;
70}

83int32_t recover_dpm_block(aaruformat_context *ctx)
84{
85 TRACE("Entering recover_dpm_block(%p)", ctx);
86
87 if(ctx == NULL || ctx->dpm_damaged_offset == 0) return AARUF_ERROR_CANNOT_READ_BLOCK;
88
89 uint8_t *recovered = NULL;
90 uint32_t size = 0;
91
92 if(ec_recover_meta_block(ctx, ctx->dpm_damaged_offset, &recovered, &size) != AARUF_STATUS_OK ||
93 recovered == NULL)
94 {
95 TRACE("Could not recover DPM block at %" PRIu64, ctx->dpm_damaged_offset);
96
97 TRACE("Exiting recover_dpm_block() = AARUF_ERROR_CANNOT_READ_BLOCK");
98 return AARUF_ERROR_CANNOT_READ_BLOCK;
99 }
100
101 int32_t status = AARUF_ERROR_CANNOT_READ_BLOCK;
102
103 if(size >= sizeof(DpmHeader))
104 {
105 DpmHeader header;
106 memcpy(&header, recovered, sizeof(DpmHeader));
107
108 const size_t block_length = sizeof(DpmHeader) + (size_t)header.length;
109
110 if(header.length <= UINT32_MAX && size >= block_length &&
111 validate_dpm_block(recovered, block_length) == AARUF_STATUS_OK)
112 {
113 uint8_t *block = malloc(block_length);
114
115 if(block != NULL)
116 {
117 TRACE("Recovered DPM block at %" PRIu64 " with erasure coding", ctx->dpm_damaged_offset);
118
119 memcpy(block, recovered, block_length);
120 free(ctx->dpm_block);
121 ctx->dpm_block = block;
122 ctx->dpm_block_length = block_length;
123 ctx->dpm_damaged_offset = 0;
124 status = AARUF_STATUS_OK;
125 }
126 }
127 }
128
129 free(recovered);
130
131 TRACE("Exiting recover_dpm_block() = %d", status);
132 return status;
133}

148void process_dpm_block(aaruformat_context *ctx, const IndexEntry *entry)
149{
150 TRACE("Entering process_dpm_block(%p, %p)", ctx, entry);
151
152 if(ctx == NULL || ctx->imageStream == NULL)
153 {
154 FATAL("Invalid context or image stream.");
155
156 TRACE("Exiting process_dpm_block()");
157 return;
158 }
159
160 if(aaruf_fseek(ctx->imageStream, (aaru_off_t)entry->offset, SEEK_SET) < 0 ||
161 aaruf_ftell(ctx->imageStream) != (aaru_off_t)entry->offset)
162 {
163 FATAL("Could not seek to %" PRIu64 " as indicated by index entry...", entry->offset);
164
165 TRACE("Exiting process_dpm_block()");
166 return;
167 }
168
169 DpmHeader header;
170
171 // From here on, a damaged block is recovered with erasure coding if the image has it
172 ctx->dpm_damaged_offset = entry->offset;
173
174 if(fread(&header, 1, sizeof(DpmHeader), ctx->imageStream) != sizeof(DpmHeader))
175 {
176 TRACE("Could not read DPM header, trying to recover it...");
177 recover_dpm_block(ctx);
178
179 TRACE("Exiting process_dpm_block()");
180 return;
181 }
182
183 if(header.identifier != DataPositionMeasurementBlock || header.length != dpm_payload_length(&header) ||
184 header.length > UINT32_MAX)
185 {
186 TRACE("Incorrect DPM header at position %" PRIu64 ", trying to recover it...", entry->offset);
187 recover_dpm_block(ctx);
188
189 TRACE("Exiting process_dpm_block()");
190 return;
191 }
192
193 const size_t block_length = sizeof(DpmHeader) + (size_t)header.length;
194 uint8_t *block = malloc(block_length);
195
196 if(block == NULL)
197 {
198 TRACE("Could not allocate memory for DPM block, continuing...");
199
200 TRACE("Exiting process_dpm_block()");
201 return;
202 }
203
204 memcpy(block, &header, sizeof(DpmHeader));
205
206 if(fread(block + sizeof(DpmHeader), 1, header.length, ctx->imageStream) != header.length ||
207 validate_dpm_block(block, block_length) != AARUF_STATUS_OK)
208 {
209 TRACE("Could not read DPM block or it is corrupted, trying to recover it...");
210 free(block);
211 recover_dpm_block(ctx);
212
213 TRACE("Exiting process_dpm_block()");
214 return;
215 }
216
217 free(ctx->dpm_block);
218 ctx->dpm_block = block;
219 ctx->dpm_block_length = block_length;
220 ctx->dpm_damaged_offset = 0;
221
222 ctx->image_info.ImageSize += header.length;
223
224 TRACE("Found DPM block at %" PRIu64 " with %u entries.", entry->offset, header.entries);
225
226 TRACE("Exiting process_dpm_block()");
227}
