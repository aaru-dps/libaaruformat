98{
99 TRACE("Entering process_ddt_v2(%p, %p, %d)", ctx, entry, *found_user_data_ddt);

153 ctx->primary_ddt_offset = entry->offset;

155 // Check for DDT compression

168 cmp_data = (uint8_t *)malloc(lzma_size);
169 if(cmp_data == NULL)
170 {
171 TRACE("Cannot allocate memory for DDT, continuing...");
172 break;
173 }
174
175 buffer = malloc(ddt_header.length);
176 if(buffer == NULL)
177 {
178 TRACE("Cannot allocate memory for DDT, continuing...");
179 free(cmp_data);
180 break;
181 }
182

184 if(read_bytes != LZMA_PROPERTIES_LENGTH)

663int32_t decode_ddt_entry_v2(aaruformat_context *ctx, const uint64_t sector_address, bool negative, uint64_t *offset,
664 uint64_t *block_offset, uint8_t *sector_status)
665{
666 TRACE("Entering decode_ddt_entry_v2(%p, %" PRIu64 ", %d, %llu, %llu, %d)", ctx, sector_address, negative, *offset,
667 *block_offset, *sector_status);
668 // Check if the context and image stream are valid
669 if(ctx == NULL || ctx->imageStream == NULL)
670 {
671 FATAL("Invalid context or image stream.");
672
673 TRACE("Exiting decode_ddt_entry_v2() = AARUF_ERROR_NOT_AARUFORMAT");
674 return AARUF_ERROR_NOT_AARUFORMAT;
675 }
676
677 if(ctx->user_data_ddt_header.tableShift > 0)
678 {
679 int32_t result = decode_ddt_multi_level_v2(ctx, sector_address, negative, offset, block_offset, sector_status);
680
681 /* If secondary DDT load failed, attempt EC recovery */
682 if(result != AARUF_STATUS_OK && ctx->ec_recovery_available && !ctx->ec_recovery_in_progress)
683 {
684 /* Compute which secondary DDT failed */
685 uint64_t sa = negative ? ctx->user_data_ddt_header.negative - sector_address
686 : sector_address + ctx->user_data_ddt_header.negative;
687 int items = 1 << ctx->user_data_ddt_header.tableShift;
688 uint64_t pos = sa / items;
689 uint64_t ddt_off = ctx->user_data_ddt2[pos];
690 ddt_off *= 1 << ctx->user_data_ddt_header.blockAlignmentShift;
691
692 TRACE("Attempting EC recovery for DDT secondary at offset %" PRIu64, ddt_off);
693 uint8_t *recovered = NULL;
694 uint32_t rec_size = 0;
695 if(ec_recover_ddt_block(ctx, ddt_off, &recovered, &rec_size) == AARUF_STATUS_OK && recovered)
696 {
697 /* Parse DdtHeader2 from recovered bytes, decompress, load into cache */
698 if(rec_size >= sizeof(DdtHeader2))
699 {
700 DdtHeader2 rec_hdr;
701 memcpy(&rec_hdr, recovered, sizeof(DdtHeader2));
702
703 uint8_t *payload = recovered + sizeof(DdtHeader2);
704 uint32_t payload_size = rec_size - sizeof(DdtHeader2);
705 uint8_t *ddt_data = NULL;
706
707 if(rec_hdr.compression == kCompressionNone && payload_size >= rec_hdr.length)
708 {
709 ddt_data = (uint8_t *)malloc((size_t)rec_hdr.length);
710 if(ddt_data) memcpy(ddt_data, payload, (size_t)rec_hdr.length);
711 }
712 else if(rec_hdr.compression == kCompressionLzma && payload_size > LZMA_PROPERTIES_LENGTH)
713 {
714 ddt_data = (uint8_t *)malloc((size_t)rec_hdr.length);
715 if(ddt_data)
716 {
717 size_t out_sz = (size_t)rec_hdr.length;
718 size_t src_sz = payload_size - LZMA_PROPERTIES_LENGTH;
719 aaruf_lzma_decode_buffer(ddt_data, &out_sz, payload + LZMA_PROPERTIES_LENGTH,
720 &src_sz, payload, LZMA_PROPERTIES_LENGTH);
721 }
722 }
723 else if(rec_hdr.compression == kCompressionZstd && payload_size > 0)
724 {
725 ddt_data = (uint8_t *)malloc((size_t)rec_hdr.length);
726 if(ddt_data)
727 aaruf_zstd_decode_buffer(ddt_data, (size_t)rec_hdr.length, payload, payload_size);
728 }
729
730 if(ddt_data)
731 {
732 TRACE("EC recovery succeeded for DDT secondary, loading into cache");
733 free(ctx->cached_secondary_ddt2);
734 ctx->cached_secondary_ddt2 = (uint64_t *)ddt_data;
735 ctx->cached_ddt_offset = ddt_off;
736 /* Retry the decode */
737 result = decode_ddt_multi_level_v2(ctx, sector_address, negative,
738 offset, block_offset, sector_status);
739 }
740 }
741 free(recovered);
742 }
743 }
744
745 return result;
746 }
747
748 return decode_ddt_single_level_v2(ctx, sector_address, negative, offset, block_offset, sector_status);

948int32_t decode_ddt_multi_level_v2(aaruformat_context *ctx, uint64_t sector_address, bool negative, uint64_t *offset,
949 uint64_t *block_offset, uint8_t *sector_status)
950{
951 TRACE("Entering decode_ddt_multi_level_v2(%p, %" PRIu64 ", %d, %llu, %llu, %d)", ctx, sector_address, negative,
952 *offset, *block_offset, *sector_status);
953
954 uint64_t ddt_entry = 0;
955 uint8_t lzma_properties[LZMA_PROPERTIES_LENGTH];
956 size_t lzma_size = 0;
957 uint8_t *cmp_data = NULL;
958 uint8_t *buffer = NULL;
959 crc64_ctx *crc64_context = NULL;
960 uint64_t crc64 = 0;
961 int items_per_ddt_entry = 0;
962 uint64_t ddt_position = 0;
963 uint64_t secondary_ddt_offset = 0;
964
965 // Check if the context and image stream are valid
966 if(ctx == NULL || ctx->imageStream == NULL)
967 {
968 FATAL("Invalid context or image stream.");
969
970 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_NOT_AARUFORMAT");
971 return AARUF_ERROR_NOT_AARUFORMAT;
972 }
973
974 // Should not really be here
975 if(ctx->user_data_ddt_header.tableShift == 0)
976 {
977 FATAL("DDT table shift is zero, but we are in multi-level DDT decoding.");
978 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
979 return AARUF_ERROR_CANNOT_READ_BLOCK;
980 }
981
982 // Calculate positive or negative sector
983 if(negative)
984 sector_address = ctx->user_data_ddt_header.negative - sector_address;
985 else
986 sector_address += ctx->user_data_ddt_header.negative;
987
988 items_per_ddt_entry = 1 << ctx->user_data_ddt_header.tableShift;
989 ddt_position = sector_address / items_per_ddt_entry;
990 secondary_ddt_offset = ctx->user_data_ddt2[ddt_position];
991
992 // Position in file of the child DDT table
993 secondary_ddt_offset *= 1 << ctx->user_data_ddt_header.blockAlignmentShift;
994
995 // Is the one we have cached the same as the one we need to read?
996 if(ctx->cached_ddt_offset != secondary_ddt_offset)
997 {
998 int32_t error_no = 0;
999 aaruf_fseek(ctx->imageStream, (aaru_off_t)secondary_ddt_offset, SEEK_SET);
1000 DdtHeader2 ddt_header;
1001 size_t read_bytes = fread(&ddt_header, 1, sizeof(DdtHeader2), ctx->imageStream);
1002
1003 if(read_bytes != sizeof(DdtHeader2))
1004 {
1005 FATAL("Could not read block header at %" PRIu64 "", secondary_ddt_offset);
1006 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1007 return AARUF_ERROR_CANNOT_READ_BLOCK;
1008 }
1009
1010 if((ddt_header.identifier != DeDuplicationTableSecondary &&
1011 ddt_header.identifier != DeDuplicationTableSAlpha) ||
1012 ddt_header.type != kDataTypeUserData)
1013 {
1014 FATAL("Invalid block header at %" PRIu64 "", secondary_ddt_offset);
1015 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1016 return AARUF_ERROR_CANNOT_READ_BLOCK;
1017 }
1018
1019 // Check for DDT compression
1020 switch(ddt_header.compression)
1021 {
1022 case kCompressionLzma:
1023 if(ddt_header.cmpLength <= LZMA_PROPERTIES_LENGTH)
1024 {
1025 FATAL("Compressed DDT payload too small (%" PRIu64 ") for LZMA properties.", ddt_header.cmpLength);
1026 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1027 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1028 }
1029
1030 lzma_size = (size_t)(ddt_header.cmpLength - LZMA_PROPERTIES_LENGTH);
1031
1032 cmp_data = (uint8_t *)malloc(lzma_size);
1033 if(cmp_data == NULL)
1034 {
1035 FATAL("Cannot allocate memory for DDT, stopping...");
1036 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1037 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1038 }
1039
1040 buffer = malloc(ddt_header.length);
1041 if(buffer == NULL)
1042 {
1043 FATAL("Cannot allocate memory for DDT, stopping...");
1044 free(cmp_data);
1045 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1046 }
1047
1048 read_bytes = fread(lzma_properties, 1, LZMA_PROPERTIES_LENGTH, ctx->imageStream);
1049 if(read_bytes != LZMA_PROPERTIES_LENGTH)
1050 {
1051 FATAL("Could not read LZMA properties, stopping...");
1052 free(cmp_data);
1053 free(buffer);
1054 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1055 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1056 }
1057
1058 read_bytes = fread(cmp_data, 1, lzma_size, ctx->imageStream);
1059 if(read_bytes != lzma_size)
1060 {
1061 FATAL("Could not read compressed block, stopping...");
1062 free(cmp_data);
1063 free(buffer);
1064 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1065 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1066 }
1067
1068 TRACE("Decompressing block of size %zu bytes", ddt_header.length);
1069 read_bytes = ddt_header.length;
1070 error_no = aaruf_lzma_decode_buffer(buffer, &read_bytes, cmp_data, &lzma_size, lzma_properties,
1071 LZMA_PROPERTIES_LENGTH);
1072
1073 if(error_no != 0)
1074 {
1075 FATAL("Got error %d from LZMA, stopping...", error_no);
1076 free(cmp_data);
1077 free(buffer);
1078 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1079 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1080 }
1081
1082 if(read_bytes != ddt_header.length)
1083 {
1084 FATAL("Error decompressing block, should be {0} bytes but got {1} bytes., stopping...");
1085 free(cmp_data);
1086 free(buffer);
1087 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1088 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1089 }
1090
1091 free(cmp_data);
1092
1093 crc64_context = aaruf_crc64_init();
1094
1095 if(crc64_context == NULL)
1096 {
1097 FATAL("Could not initialize CRC64.");
1098 free(buffer);
1099 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1100 return AARUF_ERROR_CANNOT_READ_BLOCK;
1101 }
1102
1103 aaruf_crc64_update(crc64_context, buffer, read_bytes);
1104 aaruf_crc64_final(crc64_context, &crc64);
1105 aaruf_crc64_free(crc64_context);
1106
1107 if(crc64 != ddt_header.crc64)
1108 {
1109 FATAL("Expected DDT CRC 0x%16lX but got 0x%16lX.", ddt_header.crc64, crc64);
1110 free(buffer);
1111 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_INVALID_BLOCK_CRC");
1112 return AARUF_ERROR_INVALID_BLOCK_CRC;
1113 }
1114
1115 // Free old cached DDT before replacing it
1116 free(ctx->cached_secondary_ddt2);
1117
1118 ctx->cached_secondary_ddt2 = (uint64_t *)buffer;
1119
1120 ctx->cached_ddt_offset = secondary_ddt_offset;
1121
1122 break;
1123 case kCompressionZstd:
1124 if(ddt_header.cmpLength == 0)
1125 {
1126 FATAL("Compressed DDT payload has zero length for zstd.");
1127 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1128 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1129 }
1130
1131 cmp_data = (uint8_t *)malloc(ddt_header.cmpLength);
1132 if(cmp_data == NULL)
1133 {
1134 FATAL("Cannot allocate memory for DDT, stopping...");
1135 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1136 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1137 }
1138
1139 buffer = malloc(ddt_header.length);
1140 if(buffer == NULL)
1141 {
1142 FATAL("Cannot allocate memory for DDT, stopping...");
1143 free(cmp_data);
1144 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1145 }
1146
1147 read_bytes = fread(cmp_data, 1, ddt_header.cmpLength, ctx->imageStream);
1148 if(read_bytes != ddt_header.cmpLength)
1149 {
1150 FATAL("Could not read compressed block, stopping...");
1151 free(cmp_data);
1152 free(buffer);
1153 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1154 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1155 }
1156
1157 TRACE("Decompressing block of size %zu bytes", ddt_header.length);
1158 read_bytes =
1159 aaruf_zstd_decode_buffer(buffer, ddt_header.length, cmp_data, ddt_header.cmpLength);
1160
1161 if(read_bytes != ddt_header.length)
1162 {
1163 FATAL("Error decompressing zstd DDT, expected %zu got %zu", ddt_header.length, read_bytes);
1164 free(cmp_data);
1165 free(buffer);
1166 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK");
1167 return AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1168 }
1169
1170 free(cmp_data);
1171
1172 crc64_context = aaruf_crc64_init();
1173
1174 if(crc64_context == NULL)
1175 {
1176 FATAL("Could not initialize CRC64.");
1177 free(buffer);
1178 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1179 return AARUF_ERROR_CANNOT_READ_BLOCK;
1180 }
1181
1182 aaruf_crc64_update(crc64_context, buffer, read_bytes);
1183 aaruf_crc64_final(crc64_context, &crc64);
1184 aaruf_crc64_free(crc64_context);
1185
1186 if(crc64 != ddt_header.crc64)
1187 {
1188 FATAL("Expected DDT CRC 0x%16lX but got 0x%16lX.", ddt_header.crc64, crc64);
1189 free(buffer);
1190 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_INVALID_BLOCK_CRC");
1191 return AARUF_ERROR_INVALID_BLOCK_CRC;
1192 }
1193
1194 // Free old cached DDT before replacing it
1195 free(ctx->cached_secondary_ddt2);
1196
1197 ctx->cached_secondary_ddt2 = (uint64_t *)buffer;
1198
1199 ctx->cached_ddt_offset = secondary_ddt_offset;
1200
1201 break;
1202 case kCompressionNone:
1203 buffer = malloc(ddt_header.length);
1204
1205 if(buffer == NULL)
1206 {
1207 FATAL("Cannot allocate memory for DDT, stopping...");
1208 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1209 return AARUF_ERROR_CANNOT_READ_BLOCK;
1210 }
1211
1212 read_bytes = fread(buffer, 1, ddt_header.length, ctx->imageStream);
1213
1214 if(read_bytes != ddt_header.length)
1215 {
1216 free(buffer);
1217 FATAL("Could not read deduplication table, stopping...");
1218 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1219 return AARUF_ERROR_CANNOT_READ_BLOCK;
1220 }
1221
1222 crc64_context = aaruf_crc64_init();
1223
1224 if(crc64_context == NULL)
1225 {
1226 FATAL("Could not initialize CRC64.");
1227 free(buffer);
1228 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1229 return AARUF_ERROR_CANNOT_READ_BLOCK;
1230 }
1231
1232 aaruf_crc64_update(crc64_context, buffer, read_bytes);
1233 aaruf_crc64_final(crc64_context, &crc64);
1234 aaruf_crc64_free(crc64_context);
1235
1236 if(crc64 != ddt_header.crc64)
1237 {
1238 FATAL("Expected DDT CRC 0x%16lX but got 0x%16lX.", ddt_header.crc64, crc64);
1239 free(buffer);
1240 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_INVALID_BLOCK_CRC");
1241 return AARUF_ERROR_INVALID_BLOCK_CRC;
1242 }
1243
1244 // Free old cached DDT before replacing it
1245 free(ctx->cached_secondary_ddt2);
1246
1247 ctx->cached_secondary_ddt2 = (uint64_t *)buffer;
1248
1249 ctx->cached_ddt_offset = secondary_ddt_offset;
1250
1251 break;
1252 default:
1253 FATAL("Found unknown compression type %d, stopping...", ddt_header.compression);
1254 TRACE("Exiting decode_ddt_multi_level_v2() = AARUF_ERROR_CANNOT_READ_BLOCK");
1255 return AARUF_ERROR_CANNOT_READ_BLOCK;
1256 }
1257 }
1258
1259 ddt_entry = ctx->cached_secondary_ddt2[sector_address % items_per_ddt_entry];
1260
1261 if(ddt_entry == 0)
1262 {
1263 *sector_status = SectorStatusNotDumped;
1264 *offset = 0;
1265 *block_offset = 0;
1266
1267 TRACE("Exiting decode_ddt_multi_level_v2(%p, %" PRIu64 ", %llu, %llu, %d) = AARUF_STATUS_OK", ctx,
1268 sector_address, *offset, *block_offset, *sector_status);
1269 return AARUF_STATUS_OK;
1270 }
1271
1272 *sector_status = ddt_entry >> 60;
1273 ddt_entry &= 0x0FFFFFFFFFFFFFFF;
1274
1275 const uint64_t offset_mask = (uint64_t)((1 << ctx->user_data_ddt_header.dataShift) - 1);
1276 *offset = ddt_entry & offset_mask;
1277 *block_offset =
1278 (ddt_entry >> ctx->user_data_ddt_header.dataShift) * (1 << ctx->user_data_ddt_header.blockAlignmentShift);
1279
1280 TRACE("Exiting decode_ddt_multi_level_v2(%p, %" PRIu64 ", %d, %llu, %llu, %d) = AARUF_STATUS_OK", ctx,
1281 sector_address, negative, *offset, *block_offset, *sector_status);
1282 return AARUF_STATUS_OK;
1283}

1420 crc64_ctx *crc64_context = NULL;
1421 uint64_t crc64 = 0;
1422 DdtHeader2 ddt_header;
1423 size_t written_bytes = 0;
1424 long end_of_file = 0;
1425 bool create_new_table = false;

1429 {
1430 FATAL("Invalid context or image stream.");

1456

1458 if(ctx->cached_ddt_offset == secondary_ddt_offset && secondary_ddt_offset != 0)

1461 if(*ddt_entry == 0)
1462 {

1677

1679 ctx->cached_ddt_offset = 0;
1680 ctx->cached_ddt_position = 0;
1681
1682 // Restore file position
1683 aaruf_fseek(ctx->imageStream, saved_pos, SEEK_SET);
1684
1685 TRACE("Successfully wrote never-written cached secondary DDT to disk");
1686 }
1687 else

1690 // No need to write to disk, just continue with the cached table
1691 }

1720 ddt_header.blocks = items_per_ddt_entry;

1724 ddt_header.dataShift = ctx->user_data_ddt_header.dataShift;
1725 ddt_header.tableShift = 0; // Secondary tables are single level

1734 if(crc64_context == NULL)

1743 aaruf_crc64_final(crc64_context, &crc64);

1842 {
1843 entry = (IndexEntry *)utarray_eltptr(ctx->index_entries, i);
1844 if(entry && entry->offset == ctx->cached_ddt_offset &&
1845 (entry->blockType == DeDuplicationTableSecondary || entry->blockType == DeDuplicationTableSAlpha))
1846 {
1847 TRACE("Found old DDT index entry at position %u, removing", i);
1848 utarray_erase(ctx->index_entries, i, 1);
1849 break;
1850 }
1851 }
1852 }

1855 IndexEntry new_ddt_entry;
1856 new_ddt_entry.blockType = DeDuplicationTableSecondary;
1857 new_ddt_entry.dataType = kDataTypeUserData;

1861 ctx->dirty_index_block = true;
1862 TRACE("Added new DDT index entry at offset %" PRIu64, end_of_file);

1907 create_new_table = ctx->cached_secondary_ddt2 == NULL;
1908
1909 if(!create_new_table && secondary_ddt_offset != 0)
1910 {
1911 // Load existing table
1912 aaruf_fseek(ctx->imageStream, (aaru_off_t)secondary_ddt_offset, SEEK_SET);
1913 size_t read_bytes = fread(&ddt_header, 1, sizeof(DdtHeader2), ctx->imageStream);
1914

1916 ddt_header.type != kDataTypeUserData)
1917 {
1918 FATAL("Invalid secondary DDT header at %" PRIu64, secondary_ddt_offset);
1919 TRACE("Exiting set_ddt_multi_level_v2() = false");
1920 return false;
1921 }
1922

1924 buffer = malloc(ddt_header.length);
1925 if(buffer == NULL)
1926 {
1927 FATAL("Cannot allocate memory for secondary DDT.");
1928 TRACE("Exiting set_ddt_multi_level_v2() = false");
1929 return false;
1930 }
1931
1932 read_bytes = fread(buffer, 1, ddt_header.length, ctx->imageStream);
1933 if(read_bytes != ddt_header.length)
1934 {
1935 FATAL("Could not read secondary DDT data.");
1936 free(buffer);
1937 TRACE("Exiting set_ddt_multi_level_v2() = false");
1938 return false;
1939 }

1943 if(crc64_context == NULL)
1944 {
1945 FATAL("Could not initialize CRC64.");
1946 free(buffer);
1947 TRACE("Exiting set_ddt_multi_level_v2() = false");
1948 return false;
1949 }
1950
1951 aaruf_crc64_update(crc64_context, buffer, read_bytes);
1952 aaruf_crc64_final(crc64_context, &crc64);
1953 aaruf_crc64_free(crc64_context);
1954
1955 if(crc64 != ddt_header.crc64)
1956 {
1957 FATAL("Secondary DDT CRC mismatch. Expected 0x%16lX but got 0x%16lX.", ddt_header.crc64, crc64);
1958 free(buffer);
1959 TRACE("Exiting set_ddt_multi_level_v2() = false");
1960 return false;
1961 }
1962
1963 // Cache the loaded table
1964
1965 ctx->cached_secondary_ddt2 = (uint64_t *)buffer;
1966
1967 ctx->cached_ddt_offset = secondary_ddt_offset;
1968 }
1969
1970 if(create_new_table)
1971 {
1972 // Create a new empty table
1973 size_t table_size = items_per_ddt_entry * sizeof(uint64_t);
1974
1975 buffer = calloc(1, table_size);
1976 if(buffer == NULL)
1977 {
1978 FATAL("Cannot allocate memory for new secondary DDT.");
1979 TRACE("Exiting set_ddt_multi_level_v2() = false");
1980 return false;
1981 }
1982
1983 ctx->cached_secondary_ddt2 = (uint64_t *)buffer;
1984
1985 ctx->cached_ddt_offset = 0; // Will be set when written to file
1986 ctx->cached_ddt_position = ddt_position; // Track which primary DDT position this new table belongs to
1987 TRACE("Created new secondary DDT for position %" PRIu64, ddt_position);
1988 }
1989
1990 // Step 6: Update the corresponding DDT entry
1991 if(*ddt_entry == 0)
1992 {
1993 block_index = block_offset >> ctx->user_data_ddt_header.blockAlignmentShift;
1994 *ddt_entry = offset & (1ULL << ctx->user_data_ddt_header.dataShift) - 1 |
1995 block_index << ctx->user_data_ddt_header.dataShift;
1996
1997 // Overflow detection for DDT entry
1998 if(*ddt_entry > 0xFFFFFFFFFFFFFFF)
1999 {
2000 FATAL("DDT overflow: media does not fit in big DDT");
2001 TRACE("Exiting set_ddt_multi_level_v2() = false");
2002 return false;
2003 }
2004 }
2005
2006 // Sector status can be different from previous deduplicated sector
2007 *ddt_entry &= 0x0FFFFFFFFFFFFFFF;
2008 *ddt_entry |= (uint64_t)sector_status << 60;
2009
2010 TRACE("Setting big secondary DDT entry %d to %ull", sector_address % items_per_ddt_entry, (uint64_t)*ddt_entry);
2011 ctx->cached_secondary_ddt2[sector_address % items_per_ddt_entry] = *ddt_entry;
2012 ctx->dirty_secondary_ddt = true;
2013
2014 TRACE("Updated secondary DDT entry at position %" PRIu64, sector_address % items_per_ddt_entry);
2015 TRACE("Exiting set_ddt_multi_level_v2() = true");
2016 return true;
2017}
