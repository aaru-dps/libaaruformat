182static uint8_t *twin_slot(aaruformat_context *ctx, const int comp, uint64_t index, const bool allocate)
183{
184 const uint32_t size = twin_component_size(ctx, comp);
185 uint64_t total = twin_total_sectors(ctx);
186 uint8_t **array = NULL;
187
188 // Apple / Priam tags cover the user area and overflow sectors only, indexed by sector address
189 if(comp == kTwinCompTag)
190 {
191 if(index < ctx->user_data_ddt_header.negative) return NULL;
192
193 index -= ctx->user_data_ddt_header.negative;
194 total = ctx->image_info.Sectors + ctx->user_data_ddt_header.overflow;
195 }
196
197 if(size == 0 || index >= total) return NULL;
198
199 switch(comp)
200 {
201 case kTwinCompPrefix:
202 array = (uint8_t **)&ctx->sector_prefix_ddt2;
203 break;
204 case kTwinCompSuffix:
205 array = (uint8_t **)&ctx->sector_suffix_ddt2;
206 break;
207 case kTwinCompSubheader:
208 array = &ctx->mode2_subheaders;
209 break;
210 case kTwinCompSubchannel:
211 case kTwinCompTag:
212 array = &ctx->sector_subchannel;
213 break;
214 case kTwinCompDvdId:
215 array = &ctx->sector_id;
216 break;
217 case kTwinCompDvdIed:
218 array = &ctx->sector_ied;
219 break;
220 case kTwinCompDvdCprMai:
221 array = &ctx->sector_cpr_mai;
222 break;
223 case kTwinCompSectorEdc:
224 array = &ctx->sector_edc;
225 break;
226 case kTwinCompTitleKey:
227 array = &ctx->sector_decrypted_title_key;
228 break;
229 default:
230 return NULL;
231 }
232
233 if(*array == NULL)
234 {
235 if(!allocate) return NULL;
236
237 *array = calloc(total, size);
238 if(*array == NULL) return NULL;
239 }
240
241 return *array + index * size;
242}

252static void twin_swap_in(aaruformat_context *ctx, const uint64_t index, TwinInstance *instance, TwinSlotBackup *backup)
253{
254 for(int comp = kTwinCompData + 1; comp < kTwinCompCount; comp++)
255 {
256 const uint32_t size = twin_component_size(ctx, comp);
257 const bool present = instance != NULL && (instance->present & 1U << comp) != 0;
258
259 backup->had[comp] = false;
260 if(size == 0) continue;
261
262 // Storage that did not exist before only gets created when the twin carries a value for it
263 uint8_t *existing = twin_slot(ctx, comp, index, false);
264 uint8_t *slot = existing != NULL ? existing : twin_slot(ctx, comp, index, present);
265
266 if(slot == NULL) continue;
267
268 if(existing != NULL)
269 {
270 backup->had[comp] = true;
271 memcpy(backup->bytes[comp], slot, size);
272 }
273
274 if(present)
275 memcpy(slot, twin_instance_field(instance, comp), size);
276 else
277 memset(slot, 0, size);
278 }
279}

282static void twin_capture(aaruformat_context *ctx, const uint64_t index, TwinInstance *instance)
283{
284 for(int comp = kTwinCompData + 1; comp < kTwinCompCount; comp++)
285 {
286 const uint32_t size = twin_component_size(ctx, comp);
287 uint8_t *slot = size == 0 ? NULL : twin_slot(ctx, comp, index, false);
288
289 if(slot == NULL)
290 {
291 instance->present &= ~(1U << comp);
292 continue;
293 }
294
295 memcpy(twin_instance_field(instance, comp), slot, size);
296 instance->present |= 1U << comp;
297 }
298}

301static void twin_restore(aaruformat_context *ctx, const uint64_t index, const TwinSlotBackup *backup)
302{
303 for(int comp = kTwinCompData + 1; comp < kTwinCompCount; comp++)
304 {
305 const uint32_t size = twin_component_size(ctx, comp);
306 uint8_t *slot = size == 0 ? NULL : twin_slot(ctx, comp, index, false);
307
308 if(slot == NULL) continue;
309
310 if(backup->had[comp])
311 memcpy(slot, backup->bytes[comp], size);
312 else
313 memset(slot, 0, size);
314 }

320static int32_t twin_get_raw_ddt_entry(aaruformat_context *ctx, const uint64_t sector_address, const bool negative,
321 uint64_t *entry)
322{
323 *entry = 0;
324
325 if(ctx->ddt_version == 1)
326 {
327 uint64_t offset = 0, block_offset = 0;
328 uint8_t status = SectorStatusNotDumped;
329
330 int32_t res = decode_ddt_entry_v1(ctx, sector_address, &offset, &block_offset, &status);
331 if(res != AARUF_STATUS_OK) return res;
332
333 // Only the status is meaningful for V1 images
334 *entry = status == SectorStatusNotDumped ? 0 : (uint64_t)status << 60 | 1;
335 return AARUF_STATUS_OK;
336 }
337
338 // Safe while writing too: the multi-level decoder never disturbs the table being written
339 uint64_t offset = 0, block_offset = 0;
340 uint8_t status = SectorStatusNotDumped;
341
342 const int32_t res = decode_ddt_entry_v2(ctx, sector_address, negative, &offset, &block_offset, &status);
343 if(res != AARUF_STATUS_OK) return res;
344
345 if(status == SectorStatusNotDumped && offset == 0 && block_offset == 0) return AARUF_STATUS_OK;
346
347 *entry = (offset & (1ULL << ctx->user_data_ddt_header.dataShift) - 1 |
348 block_offset >> ctx->user_data_ddt_header.blockAlignmentShift << ctx->user_data_ddt_header.dataShift) |
349 (uint64_t)status << 60;
350 return AARUF_STATUS_OK;
351}

353static int32_t twin_validate_context(void *context, const bool writing, aaruformat_context **out)
354{
355 if(context == NULL) return AARUF_ERROR_NOT_AARUFORMAT;
356
357 aaruformat_context *ctx = context;
358
359 if(ctx->magic != AARU_MAGIC) return AARUF_ERROR_NOT_AARUFORMAT;
360
361 if(writing && !ctx->is_writing) return AARUF_READ_ONLY;
362
363 if(ctx->no_user_data_ddt) return AARUF_ERROR_USER_DATA_NOT_PRESENT;
364
365 *out = ctx;
366 return AARUF_STATUS_OK;
367}

369static int32_t twin_validate_address(const aaruformat_context *ctx, const uint64_t sector_address, const bool negative)
370{
371 if(negative && sector_address > ctx->user_data_ddt_header.negative) return AARUF_ERROR_SECTOR_OUT_OF_BOUNDS;
372
373 if(!negative && sector_address > ctx->image_info.Sectors + ctx->user_data_ddt_header.overflow - 1)
374 return AARUF_ERROR_SECTOR_OUT_OF_BOUNDS;
375
376 return AARUF_STATUS_OK;
377}

421AARU_EXPORT int32_t AARU_CALL aaruf_get_sector_twin_count(const void *context, const uint64_t sector_address,
422 const bool negative, uint32_t *count)
423{
424 TRACE("Entering aaruf_get_sector_twin_count(%p, %" PRIu64 ", %d)", context, sector_address, negative);
425
426 aaruformat_context *ctx = NULL;
427 int32_t res = twin_validate_context((void *)context, false, &ctx);
428
429 if(res != AARUF_STATUS_OK) return res;
430 if(count == NULL) return AARUF_ERROR_INCORRECT_DATA_SIZE;
431
432 res = twin_validate_address(ctx, sector_address, negative);
433 if(res != AARUF_STATUS_OK) return res;
434
435 if(negative && ctx->ddt_version == 1) return AARUF_ERROR_SECTOR_OUT_OF_BOUNDS;
436
437 const TwinSectorGroupEntry *group = twin_find_group(ctx, twin_internal_index(ctx, sector_address, negative));
438
439 if(group != NULL)
440 {
441 *count = group->count;
442 TRACE("Exiting aaruf_get_sector_twin_count() = AARUF_STATUS_OK (%u)", *count);
443 return AARUF_STATUS_OK;
444 }
445
446 uint64_t entry = 0;
447 res = twin_get_raw_ddt_entry(ctx, sector_address, negative, &entry);
448 if(res != AARUF_STATUS_OK) return res;
449
450 *count = entry == 0 || entry >> 60 == SectorStatusNotDumped ? 0 : 1;
451
452 TRACE("Exiting aaruf_get_sector_twin_count() = AARUF_STATUS_OK (%u)", *count);
453 return AARUF_STATUS_OK;
454}

463static int32_t twin_read_common(void *context, const uint64_t sector_address, const bool negative,
464 const uint32_t twin_index, const TwinOperation operation, uint8_t *data,
465 uint32_t *length, uint8_t *sector_status, const int32_t tag)
466{
467 if(twin_index == 0)
468 {
469 switch(operation)
470 {
471 case kTwinOpData:
472 return aaruf_read_sector(context, sector_address, negative, data, length, sector_status);
473 case kTwinOpLong:
474 return aaruf_read_sector_long(context, sector_address, negative, data, length, sector_status);
475 default:
476 return aaruf_read_sector_tag(context, sector_address, negative, data, length, tag);
477 }
478 }
479
480 aaruformat_context *ctx = NULL;
481 int32_t res = twin_validate_context(context, false, &ctx);
482
483 if(res != AARUF_STATUS_OK) return res;
484 if(length == NULL) return AARUF_ERROR_INCORRECT_DATA_SIZE;
485 if(operation != kTwinOpTag && sector_status == NULL) return AARUF_ERROR_INCORRECT_DATA_SIZE;
486
487 res = twin_validate_address(ctx, sector_address, negative);
488 if(res != AARUF_STATUS_OK) return res;
489
490 const uint64_t index = twin_internal_index(ctx, sector_address, negative);
491 TwinSectorGroupEntry *group = twin_find_group(ctx, index);
492
493 if(group == NULL || twin_index >= group->count) return AARUF_ERROR_TWIN_INDEX_OUT_OF_RANGE;
494
495 TwinInstance *instance = &group->instances[twin_index - 1];
496 TwinSlotBackup backup;
497
498 twin_swap_in(ctx, index, instance, &backup);
499
500 ctx->twin_read_active = true;
501 ctx->twin_read_address = sector_address;
502 ctx->twin_read_negative = negative;
503 ctx->twin_read_entry = instance->present & 1U << kTwinCompData ? instance->data : 0;
504
505 switch(operation)
506 {
507 case kTwinOpData:
508 res = aaruf_read_sector(context, sector_address, negative, data, length, sector_status);
509 break;
510 case kTwinOpLong:
511 res = aaruf_read_sector_long(context, sector_address, negative, data, length, sector_status);
512 break;
513 default:
514 res = aaruf_read_sector_tag(context, sector_address, negative, data, length, tag);
515 break;
516 }
517
518 ctx->twin_read_active = false;
519
520 twin_restore(ctx, index, &backup);
521
522 return res;
523}

588static int32_t twin_create_group(aaruformat_context *ctx, const uint64_t sector_address, const bool negative,
589 const uint64_t index, TwinSectorGroupEntry **out)
590{
591 uint64_t raw = 0;
592 int32_t res = twin_get_raw_ddt_entry(ctx, sector_address, negative, &raw);
593
594 if(res != AARUF_STATUS_OK) return res;
595
596 TwinSectorGroupEntry *group = calloc(1, sizeof(TwinSectorGroupEntry));
597 if(group == NULL) return AARUF_ERROR_NOT_ENOUGH_MEMORY;
598
599 group->sector = index;
600 group->count = 1;
601 group->twin0_status = raw == 0 ? SectorStatusNotDumped : (uint8_t)(raw >> 60);
602
603 if(group->twin0_status == SectorStatusTwin) group->twin0_status = SectorStatusDumped;
604
605 uint64_t entry = raw & TWIN_ENTRY_MASK;
606 if(!set_ddt_entry_v2(ctx, sector_address, negative, 0, 0, SectorStatusTwin, &entry))
607 {
608 free(group);
609 return AARUF_ERROR_CANNOT_SET_DDT_ENTRY;
610 }
611
612 HASH_ADD(hh, ctx->twin_sectors, sector, sizeof(uint64_t), group);
613 ctx->header.featureCompatibleRo |= AARU_FEATURE_ROCOMPAT_TWIN_SECTORS;
614
615 *out = group;
616 return AARUF_STATUS_OK;
617}

619static int32_t twin_write_common(void *context, const uint64_t sector_address, const bool negative,
620 const uint32_t twin_index, const TwinOperation operation, const uint8_t *data,
621 const uint8_t sector_status, const uint32_t length, const int32_t tag)
622{
623 if(twin_index == 0)
624 {
625 switch(operation)
626 {
627 case kTwinOpData:
628 return aaruf_write_sector(context, sector_address, negative, data, sector_status, length);
629 case kTwinOpLong:
630 return aaruf_write_sector_long(context, sector_address, negative, data, sector_status, length);
631 default:
632 return aaruf_write_sector_tag(context, sector_address, negative, data, length, tag);
633 }
634 }
635
636 aaruformat_context *ctx = NULL;
637 int32_t res = twin_validate_context(context, true, &ctx);
638
639 if(res != AARUF_STATUS_OK) return res;
640
641 if(ctx->is_tape) return AARUF_ERROR_INCORRECT_MEDIA_TYPE;
642
643 res = twin_validate_address(ctx, sector_address, negative);
644 if(res != AARUF_STATUS_OK) return res;
645
646 const uint64_t index = twin_internal_index(ctx, sector_address, negative);
647 TwinSectorGroupEntry *group = twin_find_group(ctx, index);
648 const uint32_t count = group != NULL ? group->count : 1;
649
650 if(twin_index > count || twin_index >= UINT16_MAX) return AARUF_ERROR_TWIN_INDEX_OUT_OF_RANGE;
651
652
653 TwinInstance instance;
654 memset(&instance, 0, sizeof(TwinInstance));
655 if(twin_index < count) instance = group->instances[twin_index - 1];
656
657 TwinSlotBackup backup;
658 twin_swap_in(ctx, index, &instance, &backup);
659
660 ctx->twin_write_active = true;
661 ctx->twin_write_entry_set = false;
662 ctx->twin_write_entry = 0;
663
664 switch(operation)
665 {
666 case kTwinOpData:
667 res = aaruf_write_sector(context, sector_address, negative, data, sector_status, length);
668 break;
669 case kTwinOpLong:
670 res = aaruf_write_sector_long(context, sector_address, negative, data, sector_status, length);
671 break;
672 default:
673 res = aaruf_write_sector_tag(context, sector_address, negative, data, length, tag);
674 break;
675 }
676
677 ctx->twin_write_active = false;
678
679 if(res == AARUF_STATUS_OK)
680 {
681 twin_capture(ctx, index, &instance);
682
683 if(ctx->twin_write_entry_set)
684 {
685 instance.data = ctx->twin_write_entry;
686 instance.present |= 1U << kTwinCompData;
687 }
688 }
689
690 twin_restore(ctx, index, &backup);
691
692 if(res != AARUF_STATUS_OK) return res;
693
694 if(group == NULL)
695 {
696 res = twin_create_group(ctx, sector_address, negative, index, &group);
697 if(res != AARUF_STATUS_OK) return res;
698 }
699
700 if(twin_index == group->count)
701 {
702 if(group->count - 1 >= group->capacity)
703 {
704 const uint16_t new_capacity = group->capacity == 0 ? 2 : (uint16_t)(group->capacity * 2);
705 TwinInstance *grown = realloc(group->instances, new_capacity * sizeof(TwinInstance));
706
707 if(grown == NULL) return AARUF_ERROR_NOT_ENOUGH_MEMORY;
708
709 group->instances = grown;
710 group->capacity = new_capacity;
711 }
712
713 group->count++;
714 }
715
716 group->instances[twin_index - 1] = instance;
717 ctx->dirty_twin_table = true;
718
719 return AARUF_STATUS_OK;
720}

1042static int32_t twin_parse_block(const aaruformat_context *ctx, const uint8_t *raw, const size_t raw_length,
1043 TwinSectorGroupEntry **out)
1044{
1045 TwinSectorTableHeader header;
1046
1047 *out = NULL;
1048
1049 if(raw_length < sizeof(TwinSectorTableHeader)) return AARUF_ERROR_CANNOT_READ_BLOCK;
1050
1051 memcpy(&header, raw, sizeof(TwinSectorTableHeader));
1052
1053 if(header.identifier != TwinSectorTable) return AARUF_ERROR_CANNOT_READ_BLOCK;
1054
1055 if(header.cmpLength > raw_length - sizeof(TwinSectorTableHeader)) return AARUF_ERROR_CANNOT_READ_BLOCK;
1056
1057 if(header.components == 0 || header.recordSize < 8 || header.recordSize % 8 != 0 || header.length == 0 ||
1058 header.length > UINT32_MAX)
1059 return AARUF_ERROR_CANNOT_READ_BLOCK;
1060
1061 const uint8_t *cmp = raw + sizeof(TwinSectorTableHeader);
1062 size_t cmp_length = (size_t)header.cmpLength;
1063 const uint8_t *props = NULL;
1064
1065 if(header.compression == kCompressionLzma)
1066 {
1067 if(cmp_length <= LZMA_PROPERTIES_LENGTH) return AARUF_ERROR_CANNOT_READ_BLOCK;
1068
1069 props = cmp;
1070 cmp += LZMA_PROPERTIES_LENGTH;
1071 cmp_length -= LZMA_PROPERTIES_LENGTH;
1072 }
1073
1074 if(aaruf_crc64_data(cmp, cmp_length) != header.cmpCrc64) return AARUF_ERROR_INVALID_BLOCK_CRC;
1075
1076 uint8_t *payload = malloc((size_t)header.length);
1077 if(payload == NULL) return AARUF_ERROR_NOT_ENOUGH_MEMORY;
1078
1079 int32_t res = AARUF_STATUS_OK;
1080
1081 switch(header.compression)
1082 {
1083 case kCompressionNone:
1084 if(cmp_length != header.length)
1085 res = AARUF_ERROR_CANNOT_READ_BLOCK;
1086 else
1087 memcpy(payload, cmp, cmp_length);
1088 break;
1089 case kCompressionLzma:
1090 {
1091 size_t dst_size = (size_t)header.length;
1092 size_t src_size = cmp_length;
1093 if(aaruf_lzma_decode_buffer(payload, &dst_size, cmp, &src_size, props, LZMA_PROPERTIES_LENGTH) != 0 ||
1094 dst_size != header.length)
1095 res = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1096 break;
1097 }
1098 case kCompressionZstd:
1099 if(aaruf_zstd_decode_buffer(payload, (size_t)header.length, cmp, cmp_length) != header.length)
1100 res = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1101 break;
1102 default:
1103 res = AARUF_ERROR_CANNOT_DECOMPRESS_BLOCK;
1104 break;
1105 }
1106
1107 if(res == AARUF_STATUS_OK && aaruf_crc64_data(payload, (size_t)header.length) != header.crc64)
1108 res = AARUF_ERROR_INVALID_BLOCK_CRC;
1109
1110 const uint64_t descriptors_size = (uint64_t)header.components * sizeof(TwinComponentDescriptor);
1111 const uint64_t groups_size = (uint64_t)header.groups * sizeof(TwinSectorGroup);
1112 const uint64_t records_size = header.instances * header.recordSize;
1113
1114 if(res == AARUF_STATUS_OK &&
1115 (header.instances > UINT32_MAX || descriptors_size + groups_size + records_size != header.length))
1116 res = AARUF_ERROR_CANNOT_READ_BLOCK;
1117
1118 if(res != AARUF_STATUS_OK)
1119 {
1120 free(payload);
1121 return res;
1122 }
1123
1124 const TwinComponentDescriptor *descriptors = (const TwinComponentDescriptor *)payload;
1125 const uint8_t *groups_raw = payload + descriptors_size;
1126 const uint8_t *records = groups_raw + groups_size;
1127
1128 // Map descriptors to internal components and record offsets
1129 int map[256];
1130 uint32_t offsets[256];
1131 uint32_t position = 8;
1132
1133 for(uint32_t i = 0; i < header.components; i++)
1134 {
1135 map[i] = twin_component_from_descriptor(&descriptors[i]);
1136 offsets[i] = position;
1137 position += descriptors[i].size;
1138
1139 if(map[i] == kTwinCompTag && descriptors[i].size != twin_tag_size(ctx)) map[i] = -1;
1140 }
1141
1142 if(position > header.recordSize || map[0] != kTwinCompData)
1143 {
1144 free(payload);
1145 return AARUF_ERROR_CANNOT_READ_BLOCK;
1146 }
1147
1148 const uint64_t total = twin_total_sectors(ctx);
1149 TwinSectorGroupEntry *result = NULL;
1150
1151 for(uint32_t g = 0; g < header.groups && res == AARUF_STATUS_OK; g++)
1152 {
1153 TwinSectorGroup disk_group;
1154 memcpy(&disk_group, groups_raw + g * sizeof(TwinSectorGroup), sizeof(TwinSectorGroup));
1155
1156 TwinSectorGroupEntry *existing = NULL;
1157 HASH_FIND(hh, result, &disk_group.sector, sizeof(uint64_t), existing);
1158
1159 if(disk_group.count < 2 || disk_group.sector >= total || existing != NULL ||
1160 (uint64_t)disk_group.first + disk_group.count - 1 > header.instances)
1161 {
1162 res = AARUF_ERROR_CANNOT_READ_BLOCK;
1163 break;
1164 }
1165
1166 TwinSectorGroupEntry *group = calloc(1, sizeof(TwinSectorGroupEntry));
1167 if(group != NULL) group->instances = calloc(disk_group.count - 1, sizeof(TwinInstance));
1168
1169 if(group == NULL || group->instances == NULL)
1170 {
1171 twin_free_group(group);
1172 res = AARUF_ERROR_NOT_ENOUGH_MEMORY;
1173 break;
1174 }
1175
1176 group->sector = disk_group.sector;
1177 group->count = disk_group.count;
1178 group->capacity = disk_group.count - 1;
1179 group->twin0_status = disk_group.twin0Status;
1180
1181 for(uint32_t t = 0; t < (uint32_t)disk_group.count - 1; t++)
1182 {
1183 const uint8_t *record = records + (uint64_t)(disk_group.first + t) * header.recordSize;
1184 TwinInstance *inst = &group->instances[t];
1185 uint32_t present;
1186
1187 memcpy(&present, record, sizeof(uint32_t));
1188
1189 for(uint32_t i = 0; i < header.components && i < 32; i++)
1190 {
1191 if(map[i] < 0 || (present & 1U << i) == 0) continue;
1192
1193 memcpy(twin_instance_field(inst, map[i]), record + offsets[i], descriptors[i].size);
1194 inst->present |= 1U << map[i];
1195 }
1196 }
1197
1198 HASH_ADD(hh, result, sector, sizeof(uint64_t), group);
1199 }
1200
1201 free(payload);
1202
1203 if(res != AARUF_STATUS_OK)
1204 {
1205 TwinSectorGroupEntry *group = NULL;
1206 TwinSectorGroupEntry *tmp = NULL;
1207 HASH_ITER(hh, result, group, tmp)
1208 {
1209 HASH_DEL(result, group);
1210 twin_free_group(group);
1211 }
1212 return res;
1213 }
1214
1215 *out = result;
1216 return AARUF_STATUS_OK;
1217}

1303int32_t verify_twin_sector_table(aaruformat_context *ctx, const uint64_t offset)
1304{
1305 TwinSectorGroupEntry *groups = NULL;
1306 int32_t res = twin_load(ctx, offset, &groups);
1307
1308 if(res != AARUF_STATUS_OK) return res;
1309
1310 TwinSectorGroupEntry *group = NULL;
1311 TwinSectorGroupEntry *tmp = NULL;
1312 const uint64_t negs = ctx->user_data_ddt_header.negative;
1313
1314 HASH_ITER(hh, groups, group, tmp)
1315 {
1316 if(res != AARUF_STATUS_OK) break;
1317
1318 const bool negative = group->sector < negs;
1319 const uint64_t address = negative ? negs - group->sector : group->sector - negs;
1320 uint64_t raw = 0;
1321
1322 if(twin_get_raw_ddt_entry(ctx, address, negative, &raw) != AARUF_STATUS_OK || raw >> 60 != SectorStatusTwin)
1323 {
1324 FATAL("Twinned sector %" PRIu64 " is not marked as twin in the DDT", group->sector);
1325 res = AARUF_ERROR_INVALID_BLOCK_CRC;
1326 break;
1327 }
1328
1329 for(uint16_t t = 0; t + 1 < group->count; t++)
1330 {
1331 const TwinInstance *inst = &group->instances[t];
1332
1333 if(inst->present & 1U << kTwinCompPrefix && inst->prefix >> 60 == SectorStatusErrored &&
1334 ((inst->prefix & TWIN_ENTRY_MASK) + 1) * 16 > ctx->sector_prefix_length)
1335 {
1336 FATAL("Twin of sector %" PRIu64 " references a prefix outside the arena", group->sector);
1337 res = AARUF_ERROR_INVALID_BLOCK_CRC;
1338 }
1339
1340 if(inst->present & 1U << kTwinCompSuffix && inst->suffix >> 60 == SectorStatusErrored &&
1341 ((inst->suffix & TWIN_ENTRY_MASK) + 1) * 288 > ctx->sector_suffix_length)
1342 {
1343 FATAL("Twin of sector %" PRIu64 " references a suffix outside the arena", group->sector);
1344 res = AARUF_ERROR_INVALID_BLOCK_CRC;
1345 }
1346 }
1347 }
1348
1349 HASH_ITER(hh, groups, group, tmp)
1350 {
1351 HASH_DEL(groups, group);
1352 twin_free_group(group);
1353 }
1354
1355 if(res != AARUF_STATUS_OK || ctx->ddt_version != 2) return res;
1356
1357 // Every sector marked as twin must have a table entry
1358 const uint64_t total = twin_total_sectors(ctx);
1359 for(uint64_t index = 0; index < total; index++)
1360 {
1361 if(ctx->user_data_ddt_header.tableShift == 0 && ctx->user_data_ddt2 != NULL &&
1362 ctx->user_data_ddt2[index] >> 60 != SectorStatusTwin)
1363 continue;
1364
1365 const bool negative = index < negs;
1366 const uint64_t address = negative ? negs - index : index - negs;
1367
1368 // Negative sector 0 and positive sector 0 share an index; check it once as positive
1369 if(negative && address == 0) continue;
1370
1371 uint64_t raw = 0;
1372 if(twin_get_raw_ddt_entry(ctx, address, negative, &raw) != AARUF_STATUS_OK) continue;
1373
1374 if(raw >> 60 == SectorStatusTwin && twin_find_group(ctx, index) == NULL)
1375 {
1376 FATAL("Sector %" PRIu64 " is marked as twin but is not in the twin sector table", index);
1377 return AARUF_ERROR_INVALID_BLOCK_CRC;
1378 }
1379 }
1380
1381 return AARUF_STATUS_OK;
1382}

1423 }

1527 ctx->num_threads);
1528 if(cmp_size == 0) cmp_size = payload_size;

1537 }

1542 buffer = payload;
1543 cmp_size = payload_size;
1544 header.compression = kCompressionNone;
1545 }
1546 }

1560 aaruf_fseek(ctx->imageStream, position, SEEK_SET);

1566 if(ok) ok = fwrite(buffer, cmp_size, 1, ctx->imageStream) == 1;
1567
1568 if(buffer != payload) free(buffer);
1569 free(payload);
1570
1571 if(!ok)
1572 {

1574 return AARUF_ERROR_CANNOT_WRITE_BLOCK_DATA;
1575 }
1576
1577 IndexEntry index_entry;
1578 index_entry.blockType = TwinSectorTable;
1579 index_entry.dataType = kDataTypeUserData;
1580 index_entry.offset = (uint64_t)position;
1581 utarray_push_back(ctx->index_entries, &index_entry);
1582
1583 ctx->dirty_twin_table = false;
1584
1585 TRACE("Exiting write_twin_sector_table() = AARUF_STATUS_OK (%u groups, %" PRIu64 " instances at %" PRIu64 ")",
1586 group_count, instances, (uint64_t)position);
1587 return AARUF_STATUS_OK;
