104static void resize_map(hash_map_t *map, size_t new_size)
105{
106 kv_pair_t *old_table = map->table;
107 size_t old_size = map->size;
108
109 map->table = calloc(new_size, sizeof(kv_pair_t));
110 map->size = new_size;
111 map->count = 0;
112
113 for(size_t i = 0; i < old_size; i++)
114 if(old_table[i].key != 0)
115 {
116 // Re-insert
117 size_t idx = old_table[i].key % new_size;
118
119 while(map->table[idx].key != 0) idx = (idx + 1) % new_size;
120
121 map->table[idx] = old_table[i];
122 map->count++;
123 }
124
125 free(old_table);
126}

156bool insert_map(hash_map_t *map, uint64_t key, uint64_t value)
157{
158 if((double)map->count / map->size > LOAD_FACTOR) resize_map(map, map->size * 2);
159
160 size_t idx = key % map->size;
161
162 while(map->table[idx].key != 0 && map->table[idx].key != key) idx = (idx + 1) % map->size;
163
164 if(map->table[idx].key == key) return false; // Already present
165
166 map->table[idx].key = key;
167 map->table[idx].value = value;
168 map->count++;
169
170 return true;
171}
