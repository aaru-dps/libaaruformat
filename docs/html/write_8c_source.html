47static int32_t grow_arena(uint8_t **arena, size_t *length, const size_t needed, const size_t minimum)
48{
49 if(needed <= *length) return AARUF_STATUS_OK;
50
51 size_t new_length = *length * 2;
52 if(new_length < needed) new_length = needed;
53 if(new_length < minimum) new_length = minimum;
54
55 uint8_t *grown = realloc(*arena, new_length);
56 if(grown == NULL) return AARUF_ERROR_NOT_ENOUGH_MEMORY;
57
58 memset(grown + *length, 0, new_length - *length);
59 *arena = grown;
60 *length = new_length;
61
62 return AARUF_STATUS_OK;
63}

71static bool cd_header_matches_lba(const uint8_t *data, const int64_t lba)
72{
73 for(int i = 0x0C; i <= 0x0E; i++)
74 if((data[i] >> 4) > 9 || (data[i] & 0x0F) > 9) return false;
75
76 const int minute = (data[0x0C] >> 4) * 10 + (data[0x0C] & 0x0F);
77 const int second = (data[0x0D] >> 4) * 10 + (data[0x0D] & 0x0F);
78 const int frame = (data[0x0E] >> 4) * 10 + (data[0x0E] & 0x0F);
79
80 if(second > 59 || frame > 74) return false;
81
82 return (int64_t)minute * 60 * 75 + second * 75 + frame - 150 == lba;
83}

184 return AARUF_READ_ONLY;

197 {
198 FATAL("Sector address out of bounds");
199
200 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_SECTOR_OUT_OF_BOUNDS");
201 return AARUF_ERROR_SECTOR_OUT_OF_BOUNDS;
202 }

204 if(!negative && sector_address > ctx->image_info.Sectors + ctx->user_data_ddt_header.overflow - 1)
205 {

209 return AARUF_ERROR_SECTOR_OUT_OF_BOUNDS;
210 }

212 if(length > USHRT_MAX)
213 {
214 FATAL("Sector length too large");
215
216 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_INVALID_SECTOR_LENGTH");
217 return AARUF_ERROR_INVALID_SECTOR_LENGTH;
218 }
219

221
222 // Twins (instances #1..n-1) never take part in rewind detection nor in the image checksums
223 if(!ctx->rewinded && !ctx->twin_write_active)
224 {
225 // disable checksums on first encounter of decrypted/generable sectors
226 if(sector_status == SectorStatusUnencrypted || sector_status == SectorStatusGenerable)

228 TRACE("NGCW sector detected, disabling checksums");
229 ctx->rewinded = true;
230

232 if(ctx->calculating_sha1) ctx->calculating_sha1 = false;
233 if(ctx->calculating_sha256) ctx->calculating_sha256 = false;
234 if(ctx->calculating_spamsum) ctx->calculating_spamsum = false;
235 if(ctx->calculating_blake3) ctx->calculating_blake3 = false;
236 }
237 // Negative sectors are not part of the checksums, so they do not take part in rewind detection either
238 else if(negative) {}
239 else if(sector_address <= ctx->last_written_block)
240 {
241 if(sector_address == 0 && !ctx->block_zero_written)
242 ctx->block_zero_written = true;
243 else
244 {
245 TRACE("Rewinded");
246 ctx->rewinded = true;
247
248 // Disable MD5 calculation
249 if(ctx->calculating_md5) ctx->calculating_md5 = false;
250 // Disable SHA1 calculation
251 if(ctx->calculating_sha1) ctx->calculating_sha1 = false;
252 // Disable SHA256 calculation
253 if(ctx->calculating_sha256) ctx->calculating_sha256 = false;
254 // Disable SpamSum calculation
255 if(ctx->calculating_spamsum) ctx->calculating_spamsum = false;
256 // Disable BLAKE3 calculation
257 if(ctx->calculating_blake3) ctx->calculating_blake3 = false;
258 }
259 }

262 }
263

265 if(ctx->calculating_md5 && !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long &&
266 !ctx->twin_write_active)

268 // Calculate SHA1 on-the-fly if requested and sector is within user sectors (not negative or overflow)
269 if(ctx->calculating_sha1 && !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long &&
270 !ctx->twin_write_active)

272 // Calculate SHA256 on-the-fly if requested and sector is within user sectors (not negative or overflow)
273 if(ctx->calculating_sha256 && !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long &&
274 !ctx->twin_write_active)
275 aaruf_sha256_update(&ctx->sha256_context, data, length);
276 // Calculate SpamSum on-the-fly if requested and sector is within user sectors (not negative or overflow)
277 if(ctx->calculating_spamsum && !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long &&
278 !ctx->twin_write_active)
279 aaruf_spamsum_update(ctx->spamsum_context, data, length);
280 // Calculate BLAKE3 on-the-fly if requested and sector is within user sectors (not negative or overflow)
281 if(ctx->calculating_blake3 && !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long &&
282 !ctx->twin_write_active)
283 blake3_hasher_update(ctx->blake3_context, data, length);
284
285 // Mark checksum block as dirty if any checksums are being calculated
286 if((ctx->calculating_md5 || ctx->calculating_sha1 || ctx->calculating_sha256 || ctx->calculating_spamsum ||
287 ctx->calculating_blake3) &&
288 !negative && sector_address <= ctx->image_info.Sectors && !ctx->writing_long && !ctx->twin_write_active)
289 ctx->dirty_checksum_block = true;
290
291 // PS3 decryption: if caller sends encrypted data marked as "to be stored decrypted"
292 // Checksums have already been computed on the ciphertext above.
293 const uint8_t *write_data = data;
294 uint8_t *decrypted_buffer = NULL;
295
296 if(sector_status == SectorStatusUnencrypted && (ctx->header.mediaType == PS3DVD || ctx->header.mediaType == PS3BD))
297 {
298 if(!ctx->ps3_encryption_initialized)
299 {
300 ps3_lazy_init(ctx);
301 ctx->ps3_encryption_initialized = true;
302 }
303
304 if(ctx->ps3_disc_key != NULL && ctx->ps3_plaintext_regions != NULL &&
305 ps3_is_sector_encrypted((const Ps3PlaintextRegion *)ctx->ps3_plaintext_regions,
306 ctx->ps3_plaintext_region_count, sector_address))
307 {
308 decrypted_buffer = (uint8_t *)malloc(length);
309
310 if(decrypted_buffer == NULL)
311 {
312 FATAL("Could not allocate memory for PS3 decryption buffer");
313 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
314 }
315
316 memcpy(decrypted_buffer, data, length);
317 ps3_decrypt_sector(ctx->ps3_disc_key, sector_address, decrypted_buffer, length);
318 write_data = decrypted_buffer;
319 }
320 }
321
322 // Twin #0 of a twinned sector keeps SectorStatusTwin in the DDT, its real status goes to the twin table
323 uint8_t ddt_status = sector_status;
324
325 if(!ctx->twin_write_active && ctx->twin_sectors != NULL)
326 {
327 TwinSectorGroupEntry *twin_group = twin_find_group(ctx, twin_internal_index(ctx, sector_address, negative));
328
329 if(twin_group != NULL)
330 {
331 twin_group->twin0_status = sector_status;
332 ddt_status = SectorStatusTwin;
333 ctx->dirty_twin_table = true;
334 }
335 }
336
337 // NGCW: SectorStatusGenerable — set DDT entry only, do not store data in any block
338 if(sector_status == SectorStatusGenerable && (ctx->header.mediaType == GOD || ctx->header.mediaType == WOD))
339 {
340 if(ctx->twin_write_active)
341 {
342 free(decrypted_buffer);
343 ctx->twin_write_entry = (uint64_t)sector_status << 60;
344 ctx->twin_write_entry_set = true;
345 return AARUF_STATUS_OK;
346 }
347
348 uint64_t ddt_entry_gen = 0;
349 bool ddt_ok_gen = set_ddt_entry_v2(ctx, sector_address, negative, 0, 0, ddt_status, &ddt_entry_gen);
350
351 free(decrypted_buffer);
352
353 if(!ddt_ok_gen)
354 {
355 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_CANNOT_SET_DDT_ENTRY");
356 return AARUF_ERROR_CANNOT_SET_DDT_ENTRY;
357 }
358
359 TRACE("Exiting aaruf_write_sector() = AARUF_STATUS_OK (generable)");
360 return AARUF_STATUS_OK;
361 }
362
363 // Close current block first
364 if(ctx->writing_buffer != NULL &&
365 // When sector size changes or block reaches maximum size
366 (ctx->current_block_header.sectorSize != length ||
367 ctx->current_block_offset == 1 << ctx->user_data_ddt_header.dataShift))
368 {
369 TRACE("Closing current block before writing new data");
370 int error = aaruf_close_current_block(ctx);
371
372 if(error != AARUF_STATUS_OK)
373 {
374 FATAL("Error closing current block: %d", error);
375
376 TRACE("Exiting aaruf_write_sector() = %d", error);
377 return error;
378 }
379 }
380
381 uint64_t ddt_entry = 0;
382 bool ddt_ok;
383
384 if(ctx->twin_write_active)
385 {
386 // Twin instance: produce the pointer for the twin table, the user-data DDT is left untouched
387 bool existing = false;
388 uint64_t hash = 0;
389
390 if(ctx->deduplicate)
391 {
392 hash = XXH3_64bits(write_data, length);
393 existing = lookup_map(ctx->sector_hash_map, hash, &ddt_entry);
394 }
395
396 if(!existing)
397 {
398 const uint64_t block_index = ctx->next_block_position >> ctx->user_data_ddt_header.blockAlignmentShift;
399 ddt_entry = ctx->current_block_offset & (1ULL << ctx->user_data_ddt_header.dataShift) - 1 |
400 block_index << ctx->user_data_ddt_header.dataShift;
401
402 if(ddt_entry > 0xFFFFFFFFFFFFFFF)
403 {
404 FATAL("DDT overflow: media does not fit in big DDT");
405 free(decrypted_buffer);
406 return AARUF_ERROR_CANNOT_SET_DDT_ENTRY;
407 }
408
409 if(ctx->deduplicate) insert_map(ctx->sector_hash_map, hash, ddt_entry);
410 }
411
412 ctx->twin_write_entry = ddt_entry & 0x0FFFFFFFFFFFFFFF | (uint64_t)sector_status << 60;
413 ctx->twin_write_entry_set = true;
414
415 if(existing)
416 {
417 free(decrypted_buffer);
418 TRACE("Exiting aaruf_write_sector() = AARUF_STATUS_OK (deduplicated twin)");
419 return AARUF_STATUS_OK;
420 }
421
422 ddt_ok = true;
423 }
424 else if(ctx->deduplicate)
425 {
426 // Calculate 64-bit XXH3 hash of the sector (on decrypted data for dedup)
427 TRACE("Hashing sector data for deduplication");
428 uint64_t hash = XXH3_64bits(write_data, length);
429
430 // Check if the hash is already in the map
431 bool existing = lookup_map(ctx->sector_hash_map, hash, &ddt_entry);
432 TRACE("Block does %s exist in deduplication map", existing ? "already" : "not yet");
433
434 ddt_ok = set_ddt_entry_v2(ctx, sector_address, negative, ctx->current_block_offset, ctx->next_block_position,
435 ddt_status, &ddt_entry);
436 if(!ddt_ok)
437 {
438 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_CANNOT_SET_DDT_ENTRY");
439 return AARUF_ERROR_CANNOT_SET_DDT_ENTRY;
440 }
441
442 if(existing)
443 {
444 TRACE("Sector exists, so not writing to image");
445 free(decrypted_buffer);
446 TRACE("Exiting aaruf_write_sector() = AARUF_STATUS_OK");
447 return AARUF_STATUS_OK;
448 }
449
450 TRACE("Inserting sector hash into deduplication map, proceeding to write into image as normal");
451 insert_map(ctx->sector_hash_map, hash, ddt_entry);
452 }
453 else
454 ddt_ok = set_ddt_entry_v2(ctx, sector_address, negative, ctx->current_block_offset, ctx->next_block_position,
455 ddt_status, &ddt_entry);
456
457 if(!ddt_ok)
458 {
459 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_CANNOT_SET_DDT_ENTRY");
460 return AARUF_ERROR_CANNOT_SET_DDT_ENTRY;
461 }
462
463 // No block set
464 if(ctx->writing_buffer_position == 0)
465 {
466 TRACE("Creating new writing block");
467 ctx->current_block_header.identifier = DataBlock;
468 ctx->current_block_header.type = kDataTypeUserData;
469 ctx->current_block_header.sectorSize = length;
470
471 // We need to save the track type for later compression
472 if(ctx->image_info.MetadataMediaType == OpticalDisc && ctx->track_entries != NULL)
473 {
474 TrackEntry found_track;
475 const TrackEntry *track = NULL;
476
477 // Lead-In and Lead-Out sectors take the format of the track after and before them
478 if(find_cd_track_for_sector(ctx->track_entries, ctx->tracks_header.entries,
479 negative ? -(int64_t)sector_address : (int64_t)sector_address, &found_track,
480 NULL))
481 track = &found_track;
482
483 if(track != NULL)
484 ctx->current_track_type = track->type;
485 else
486 ctx->current_track_type = kTrackTypeData;
487
488 if(ctx->current_track_type == kTrackTypeAudio &&
489 // JaguarCD stores data in audio tracks. FLAC is too inefficient, we need to use LZMA as data.
490 (ctx->image_info.MediaType == JaguarCD && track->session > 1 ||
491 // VideoNow stores video in audio tracks, and LZMA works better too.
492 ctx->image_info.MediaType == VideoNow || ctx->image_info.MediaType == VideoNowColor ||
493 ctx->image_info.MediaType == VideoNowXp))
494 ctx->current_track_type = kTrackTypeData;
495
496 if(ctx->compression_enabled)
497 {
498 if(ctx->current_track_type == kTrackTypeAudio)
499 ctx->current_block_header.compression = kCompressionFlac;
500 else
501 ctx->current_block_header.compression = ctx->use_zstd ? kCompressionZstd : kCompressionLzma;
502 }
503 else
504 ctx->current_block_header.compression = kCompressionNone;
505 }
506 else
507 {
508 ctx->current_track_type = kTrackTypeData;
509 if(ctx->compression_enabled)
510 ctx->current_block_header.compression = ctx->use_zstd ? kCompressionZstd : kCompressionLzma;
511 else
512 ctx->current_block_header.compression = kCompressionNone;
513 }
514
515 uint32_t max_buffer_size =
516 (1 << ctx->user_data_ddt_header.dataShift) * ctx->current_block_header.sectorSize * 2;
517 TRACE("Setting max buffer size to %u bytes", max_buffer_size);
518
519 TRACE("Allocating memory for writing buffer");
520 ctx->writing_buffer = (uint8_t *)calloc(1, max_buffer_size);
521 if(ctx->writing_buffer == NULL)
522 {
523 FATAL("Could not allocate memory");
524
525 TRACE("Exiting aaruf_write_sector() = AARUF_ERROR_NOT_ENOUGH_MEMORY");
526 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
527 }
528 }
529
530 TRACE("Copying data to writing buffer at position %zu", ctx->writing_buffer_position);
531 memcpy(ctx->writing_buffer + ctx->writing_buffer_position, write_data, length);
532 TRACE("Advancing writing buffer position to %zu", ctx->writing_buffer_position + length);
533 ctx->writing_buffer_position += length;
534 TRACE("Advancing current block offset to %zu", ctx->current_block_offset + 1);
535 ctx->current_block_offset++;
536
537 free(decrypted_buffer);
538
539 TRACE("Exiting aaruf_write_sector() = AARUF_STATUS_OK");
540 return AARUF_STATUS_OK;
541}

795 FATAL("Sector address out of bounds");

809 switch(ctx->image_info.MetadataMediaType)
810 {
811 case OpticalDisc:
812 {
813 TrackEntry track = {0};
814
815 // LBA as encoded in a CD sector header (lead-in sectors are negative)
816 const int64_t signed_lba = negative ? -(int64_t)sector_address : (int64_t)sector_address;
817
818 // Lead-In and Lead-Out sectors take the format of the track after and before them
819 if(!find_cd_track_for_sector(ctx->track_entries, ctx->tracks_header.entries, signed_lba, &track, NULL))
820 track.type = kTrackTypeData;

823

826 corrected_sector_address = ctx->user_data_ddt_header.negative - sector_address;
827 else
828 corrected_sector_address += ctx->user_data_ddt_header.negative;
829
830 uint64_t total_sectors =
831 ctx->user_data_ddt_header.negative + ctx->image_info.Sectors + ctx->user_data_ddt_header.overflow;
832
833 // DVD long sector
834 if(length == 2064 && (ctx->image_info.MediaType == DVDROM || ctx->image_info.MediaType == PS2DVD ||
835 ctx->image_info.MediaType == SACD || ctx->image_info.MediaType == PS3DVD ||
836 ctx->image_info.MediaType == DVDR || ctx->image_info.MediaType == DVDRW ||
837 ctx->image_info.MediaType == DVDPR || ctx->image_info.MediaType == DVDPRW ||

860 if(ctx->sector_ied == NULL) ctx->sector_ied = calloc(1, 2 * total_sectors);
861 if(ctx->sector_cpr_mai == NULL) ctx->sector_cpr_mai = calloc(1, 6 * total_sectors);

1417 }
1418

1428 }

1438

1440

1442 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
1443 }
1444 }
1445

1447 ctx->dirty_sector_subchannel_block = true; // Mark subchannel block as dirty
1448 free(newTag);

1461 return AARUF_ERROR_INCORRECT_MEDIA_TYPE;

1646int32_t aaruf_close_current_block(aaruformat_context *ctx)
1647{
1648 // Not a libaaruformat context
1649 if(ctx->magic != AARU_MAGIC) return AARUF_ERROR_NOT_AARUFORMAT;
1650
1651 // Check we are writing
1652 if(!ctx->is_writing) return AARUF_READ_ONLY;
1653
1654 ctx->current_block_header.length = ctx->current_block_offset * ctx->current_block_header.sectorSize;
1655
1656 TRACE("Initializing CRC64 context");
1657 ctx->crc64_context = aaruf_crc64_init();
1658 TRACE("Updating CRC64");
1659 aaruf_crc64_update(ctx->crc64_context, ctx->writing_buffer, ctx->current_block_header.length);
1660 aaruf_crc64_final(ctx->crc64_context, &ctx->current_block_header.crc64);
1661
1662 uint8_t lzma_properties[LZMA_PROPERTIES_LENGTH] = {0};
1663 uint8_t *cmp_buffer = NULL;
1664
1665 switch(ctx->current_block_header.compression)
1666 {
1667 case kCompressionNone:
1668 break;
1669 case kCompressionFlac:
1670 cmp_buffer = malloc(ctx->current_block_header.length * 2);
1671 if(cmp_buffer == NULL)
1672 {
1673 FATAL("Could not allocate buffer for compressed data");
1674 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
1675 }
1676 const uint32_t current_samples = ctx->current_block_offset * SAMPLES_PER_SECTOR;
1677 uint32_t flac_block_size = ctx->current_block_offset * SAMPLES_PER_SECTOR;
1678
1679 if(flac_block_size > MAX_FLAKE_BLOCK) flac_block_size = MAX_FLAKE_BLOCK;
1680 if(flac_block_size < MIN_FLAKE_BLOCK) flac_block_size = MIN_FLAKE_BLOCK;
1681
1682 const long remaining = current_samples % flac_block_size;
1683
1684 // Fill FLAC block
1685 if(remaining != 0)
1686 for(int r = 0; r < remaining * 4; r++) ctx->writing_buffer[ctx->writing_buffer_position + r] = 0;
1687
1688 ctx->current_block_header.cmpLength = aaruf_flac_encode_redbook_buffer(
1689 cmp_buffer, ctx->current_block_header.length * 2, ctx->writing_buffer, ctx->current_block_header.length,
1690 flac_block_size, true, false, "hamming", 12, 15, true, false, 0, 8, "Aaru", 4);
1691
1692 if(ctx->current_block_header.cmpLength >= ctx->current_block_header.length)
1693 {
1694 ctx->current_block_header.compression = kCompressionNone;
1695 free(cmp_buffer);
1696 }
1697
1698 break;
1699 case kCompressionLzma:
1700 cmp_buffer = malloc(ctx->current_block_header.length * 2);
1701 if(cmp_buffer == NULL)
1702 {
1703 FATAL("Could not allocate buffer for compressed data");
1704 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
1705 }
1706
1707 size_t dst_size = ctx->current_block_header.length * 2;
1708 size_t props_size = LZMA_PROPERTIES_LENGTH;
1709 aaruf_lzma_encode_buffer(cmp_buffer, &dst_size, ctx->writing_buffer, ctx->current_block_header.length,
1710 lzma_properties, &props_size, 9, ctx->lzma_dict_size, 4, 0, 2, 273, LZMA_THREADS(ctx));
1711
1712 ctx->current_block_header.cmpLength = (uint32_t)dst_size;
1713
1714 if(ctx->current_block_header.cmpLength >= ctx->current_block_header.length)
1715 {
1716 ctx->current_block_header.compression = kCompressionNone;
1717 free(cmp_buffer);
1718 }
1719
1720 break;
1721 case kCompressionZstd:
1722 cmp_buffer = malloc(ctx->current_block_header.length * 2);
1723 if(cmp_buffer == NULL)
1724 {
1725 FATAL("Could not allocate buffer for compressed data");
1726 return AARUF_ERROR_NOT_ENOUGH_MEMORY;
1727 }
1728
1729 size_t zstd_dst_size =
1730 aaruf_zstd_encode_buffer(cmp_buffer, ctx->current_block_header.length * 2, ctx->writing_buffer,
1731 ctx->current_block_header.length, ctx->zstd_level, ctx->num_threads);
1732
1733 if(zstd_dst_size == 0)
1734 {
1735 ctx->current_block_header.compression = kCompressionNone;
1736 free(cmp_buffer);
1737 cmp_buffer = NULL;
1738 }
1739 else
1740 {
1741 ctx->current_block_header.cmpLength = (uint32_t)zstd_dst_size;
1742
1743 if(ctx->current_block_header.cmpLength >= ctx->current_block_header.length)
1744 {
1745 ctx->current_block_header.compression = kCompressionNone;
1746 free(cmp_buffer);
1747 cmp_buffer = NULL;
1748 }
1749 else
1750 ctx->has_zstd_blocks = true;
1751 }
1752
1753 break;
1754 default:
1755 FATAL("Invalid compression type");
1756 return AARUF_ERROR_CANNOT_WRITE_BLOCK_DATA;
1757 }
1758
1759 if(ctx->current_block_header.compression == kCompressionNone)
1760 {
1761 ctx->current_block_header.cmpCrc64 = ctx->current_block_header.crc64;
1762 ctx->current_block_header.cmpLength = ctx->current_block_header.length;
1763 }
1764 else
1765 ctx->current_block_header.cmpCrc64 = aaruf_crc64_data(cmp_buffer, ctx->current_block_header.cmpLength);
1766
1767 if(ctx->current_block_header.compression == kCompressionLzma)
1768 ctx->current_block_header.cmpLength += LZMA_PROPERTIES_LENGTH;
1769
1770 // Add to index
1771 TRACE("Adding block to index");
1772 IndexEntry index_entry;
1773 index_entry.blockType = DataBlock;
1774 index_entry.dataType = kDataTypeUserData;
1775 index_entry.offset = ctx->next_block_position;
1776
1777 utarray_push_back(ctx->index_entries, &index_entry);
1778 ctx->dirty_index_block = true; // Mark index block as dirty
1779 TRACE("Block added to index at offset %" PRIu64, index_entry.offset);
1780
1781 // Write block header to file
1782
1783 // Move to expected block position
1784 aaruf_fseek(ctx->imageStream, (aaru_off_t)ctx->next_block_position, SEEK_SET);
1785
1786 // Write block header
1787 if(fwrite(&ctx->current_block_header, sizeof(BlockHeader), 1, ctx->imageStream) != 1)
1788 return AARUF_ERROR_CANNOT_WRITE_BLOCK_HEADER;
1789
1790 // Write block data
1791 if(ctx->current_block_header.compression == kCompressionLzma &&
1792 fwrite(lzma_properties, LZMA_PROPERTIES_LENGTH, 1, ctx->imageStream) != 1)
1793 {
1794 free(cmp_buffer);
1795 return AARUF_ERROR_CANNOT_WRITE_BLOCK_DATA;
1796 }
1797
1798 if(ctx->current_block_header.compression == kCompressionNone)
1799 {
1800 if(fwrite(ctx->writing_buffer, ctx->current_block_header.length, 1, ctx->imageStream) != 1)
1801 return AARUF_ERROR_CANNOT_WRITE_BLOCK_DATA;
1802 }
1803 else
1804 {
1805 if(fwrite(cmp_buffer, ctx->current_block_header.cmpLength, 1, ctx->imageStream) != 1)
1806 {
1807 free(cmp_buffer);
1808 return AARUF_ERROR_CANNOT_WRITE_BLOCK_DATA;
1809 }
1810 }
1811
1812 /* --- Erasure coding parity accumulation ---
1813 * Call BEFORE freeing cmp_buffer / writing_buffer.
1814 * For LZMA: cmpLength already includes LZMA_PROPERTIES_LENGTH, but the properties
1815 * were written separately. We pass them as lzma_props parameter.
1816 * For non-LZMA compressed: cmp_buffer holds all payload.
1817 * For uncompressed: writing_buffer holds payload. */
1818 if(ctx->ec_enabled)
1819 {
1820 const uint8_t *ec_payload;
1821 uint32_t ec_payload_size;
1822 const uint8_t *ec_lzma_props = NULL;
1823
1824 if(ctx->current_block_header.compression == kCompressionNone)
1825 {
1826 ec_payload = ctx->writing_buffer;
1827 ec_payload_size = ctx->current_block_header.length;
1828 }
1829 else if(ctx->current_block_header.compression == kCompressionLzma)
1830 {
1831 ec_payload = cmp_buffer;
1832 ec_payload_size = ctx->current_block_header.cmpLength - LZMA_PROPERTIES_LENGTH;
1833 ec_lzma_props = lzma_properties;
1834 }
1835 else
1836 {
1837 ec_payload = cmp_buffer;
1838 ec_payload_size = ctx->current_block_header.cmpLength;
1839 }
1840
1841 ec_accumulate_data_block(ctx, &ctx->current_block_header, ec_lzma_props,
1842 ec_payload, ec_payload_size, index_entry.offset);
1843 }
1844
1845 /* Free compressed buffer (if compression was used) */
1846 if(ctx->current_block_header.compression != kCompressionNone)
1847 free(cmp_buffer);
1848
1849 // Update nextBlockPosition to point to the next available aligned position
1850 const uint64_t block_total_size = sizeof(BlockHeader) + ctx->current_block_header.cmpLength;
1851 const uint64_t alignment_mask = (1ULL << ctx->user_data_ddt_header.blockAlignmentShift) - 1;
1852 ctx->next_block_position = ctx->next_block_position + block_total_size + alignment_mask & ~alignment_mask;
1853 TRACE("Updated nextBlockPosition to %" PRIu64, ctx->next_block_position);
1854
1855 // Clear values
1856 free(ctx->writing_buffer);
1857 ctx->writing_buffer = NULL;
1858 ctx->current_block_offset = 0;
1859 memset(&ctx->current_block_header, 0, sizeof(BlockHeader));
1860 aaruf_crc64_free(ctx->crc64_context);
1861 ctx->writing_buffer_position = 0;
1862
1863 return AARUF_STATUS_OK;
1864}
